{{- /* responsive-image.html: Generates a element with AVIF/WebP responsive srcset. */ -}} {{- $imagePath := .imagePath -}} {{- $alt := .alt -}} {{- $class := .class | default "" -}} {{- $loading := .loading | default "lazy" -}} {{- $sizes := .sizes | default "100vw" -}} {{- $widths := .widths | default (slice 400 800 1200) -}} {{- $src := resources.Get $imagePath -}} {{- if $src -}} {{- /* AVIF srcset */ -}} {{- $avifSrcset := slice -}} {{- range $widths -}} {{- $resized := $src.Resize (printf "%dx" .) -}} {{- $avif := $resized.Process "avif" -}} {{- $avifSrcset = $avifSrcset | append (printf "%s %dw" $avif.RelPermalink .) -}} {{- end -}} {{- /* WebP srcset */ -}} {{- $webpSrcset := slice -}} {{- range $widths -}} {{- $resized := $src.Resize (printf "%dx" .) -}} {{- $webp := $resized.Process "webp" -}} {{- $webpSrcset = $webpSrcset | append (printf "%s %dw" $webp.RelPermalink .) -}} {{- end -}} {{- /* Fallback : largest width in PNG for universal compatibility */ -}} {{- $fallbackWidth := index $widths (sub (len $widths) 1) -}} {{- $fallback := $src.Resize (printf "%dx" $fallbackWidth) -}} {{- else -}} {{- /* Image not found in assets — fall back to static URL */ -}} {{- end -}}
